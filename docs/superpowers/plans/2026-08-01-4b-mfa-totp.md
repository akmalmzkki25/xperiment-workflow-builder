# 4B — MFA TOTP (AUTH-004) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A TOTP second factor any user may enable, with ten single-use recovery codes and a workspace policy an owner may switch on to require it of privileged members.

**Architecture:** RFC 6238 is implemented by hand in `packages/db/src/totp.ts` and proven against the RFC's own test vectors. The secret is wrapped by the deployment's root key — not the workspace `SecretVault`, because MFA is account-level. Login becomes two steps: a password answers with a five-minute `mfa_challenge` row in the existing `auth_tokens` table instead of a session, and `POST /v1/auth/mfa/verify` trades that challenge plus a code for the session.

**Tech Stack:** pnpm monorepo, ESM, tsx (runs TS source — there is no build step), TypeScript strict with `noUncheckedIndexedAccess`, Vitest 2, Fastify 5 (api, :3000), Next.js 14 app router (web, :3005), Playwright, Postgres.

## Global Constraints

- Run the full suite with `pnpm test`. **Never `pnpm -r test`** — the root script pins `--workspace-concurrency=1`, and concurrent packages make `@xperiment/queue`'s outbox publisher race others over `task_outbox`.
- Coverage thresholds are 80% on lines, statements, functions and branches, per package.
- Conventional commit subjects, `<type>: <description>`, **no attribution trailer**.
- No `console.log`. The api logs through `req.log` / `app.log`.
- Never log or return a raw TOTP secret, base32 string, `otpauth://` URI, recovery code, or submitted code. Audit metadata is a fixed set of named scalars.
- node-postgres returns `BIGINT` as a **string**. `last_used_step` is therefore compared in SQL, never in JS.
- Fastify has **no `setErrorHandler`** in this app. A thrown error becomes a 500 carrying its message, so routes return problems explicitly.
- An unexempted public route is a dead route (the 3O payments-webhook trap). `EXEMPT` lives in `apps/api/src/auth/plugin.ts`.

```ts
import { createHmac, randomBytes, timingSafeEqual } from 'node:crypto';

/** RFC 4648 base32. '0', '1', '8' and 'I' are absent by design: they misread as O, l and B. */
const ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';

export const TOTP_STEP_SECONDS = 30;
const DEFAULT_DIGITS = 6;
const DEFAULT_WINDOW = 1;
const SECRET_BYTES = 20;

export function base32Encode(buf: Buffer): string {
  let bits = 0;
  let value = 0;
  let out = '';
  for (const byte of buf) {
    value = (value << 8) | byte;
    bits += 8;
    while (bits >= 5) {
      out += ALPHABET[(value >>> (bits - 5)) & 31];
      bits -= 5;
    }
  }
  // Left-over bits are padded with zeros rather than '='; every authenticator accepts this.
  if (bits > 0) out += ALPHABET[(value << (5 - bits)) & 31];
  return out;
}

export function base32Decode(s: string): Buffer {
  const clean = s.replace(/[\s-]/g, '').replace(/=+$/, '').toUpperCase();
  let bits = 0;
  let value = 0;
  const out: number[] = [];
  for (const ch of clean) {
    const idx = ALPHABET.indexOf(ch);
    if (idx < 0) throw new Error(`invalid base32 character: ${JSON.stringify(ch)}`);
    value = (value << 5) | idx;
    bits += 5;
    if (bits >= 8) {
      out.push((value >>> (bits - 8)) & 255);
      bits -= 8;
    }
  }
  return Buffer.from(out);
}

export function generateTotpSecret(): { secret: Buffer; base32: string } {
  const secret = randomBytes(SECRET_BYTES);
  return { secret, base32: base32Encode(secret) };
}

/**
 * One HOTP value for a counter, per RFC 4226's dynamic truncation. `digits` is a parameter
 * only so the RFC 6238 test vectors — which are eight digits — can be asserted directly;
 * everything in the product uses the six-digit default.
 */
export function totpCode(secret: Buffer, counter: number, digits: number = DEFAULT_DIGITS): string {
  const message = Buffer.alloc(8);
  message.writeBigUInt64BE(BigInt(counter));
  const mac = createHmac('sha1', secret).update(message).digest();
  const offset = mac[mac.length - 1]! & 0x0f;
  const binary =
    ((mac[offset]! & 0x7f) << 24) |
    ((mac[offset + 1]! & 0xff) << 16) |
    ((mac[offset + 2]! & 0xff) << 8) |
    (mac[offset + 3]! & 0xff);
  return String(binary % 10 ** digits).padStart(digits, '0');
}

function constantTimeEquals(a: string, b: string): boolean {
  const left = Buffer.from(a, 'utf8');
  const right = Buffer.from(b, 'utf8');
  return left.length === right.length && timingSafeEqual(left, right);
}

/**
 * Returns the step the code matched, or null. `after` is a floor: no step at or below it is
 * ever accepted, which is what keeps the ±1 drift window from doubling as a replay window.
 */
export function verifyTotpCode(args: {
  secret: Buffer;
  code: string;
  now?: number;
  window?: number;
  digits?: number;
  after?: number | null;
}): number | null {
  const digits = args.digits ?? DEFAULT_DIGITS;
  if (args.code.length !== digits || !/^\d+$/.test(args.code)) return null;
  const current = Math.floor((args.now ?? Date.now()) / 1000 / TOTP_STEP_SECONDS);
  const window = args.window ?? DEFAULT_WINDOW;
  for (let delta = -window; delta <= window; delta += 1) {
    const step = current + delta;
    if (args.after != null && step <= args.after) continue;
    if (constantTimeEquals(totpCode(args.secret, step, digits), args.code)) return step;
  }
  return null;
}

export function otpauthUri(args: { issuer: string; account: string; base32: string }): string {
  const label = encodeURIComponent(`${args.issuer}:${args.account}`);
  const params = new URLSearchParams({
    secret: args.base32,
    issuer: args.issuer,
    algorithm: 'SHA1',
    digits: String(DEFAULT_DIGITS),
    period: String(TOTP_STEP_SECONDS),
  });
  return `otpauth://totp/${label}?${params.toString()}`;
}
```

```sql
-- AUTH-004. The secret is account-level, so it is wrapped by the deployment root key
-- rather than a workspace KEK: filing it under one workspace would mean destroying that
-- workspace's key locks the user out of every other workspace they belong to.
CREATE TABLE user_mfa (
  user_id        UUID PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
  secret_enc     BYTEA NOT NULL,
  -- Which root key wrapped it. A deployment pointed at the wrong key then fails loudly
  -- instead of silently rejecting every correct code.
  root_key_id    TEXT NOT NULL,
  -- NULL means enrolment was started but never proven. Such a row is not a second factor
  -- and no login consults it.
  confirmed_at   TIMESTAMPTZ NULL,
  -- Highest TOTP step already spent. The verification window is +/-1 step, so a code is
  -- live for 90 seconds; without this floor, a code seen once can be spent again inside it.
  last_used_step BIGINT NULL,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE mfa_recovery_codes (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  -- Hashed with the same sha256 as sessions, api keys and auth tokens. The raw code is
  -- shown to the user exactly once and never stored.
  code_hash   TEXT NOT NULL UNIQUE,
  consumed_at TIMESTAMPTZ NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX mfa_recovery_codes_user_idx ON mfa_recovery_codes (user_id);

-- Owner-set policy. Enforced against sessions only; see the 4B design for why API keys
-- are deliberately outside it.
ALTER TABLE workspaces ADD COLUMN require_mfa BOOLEAN NOT NULL DEFAULT false;

-- 0013 pinned the purpose CHECK to two values. The login challenge is a third, and this
-- constraint would otherwise reject every one of them at runtime rather than at migrate
-- time. Postgres named the inline constraint auth_tokens_purpose_check.
ALTER TABLE auth_tokens DROP CONSTRAINT auth_tokens_purpose_check;
ALTER TABLE auth_tokens ADD CONSTRAINT auth_tokens_purpose_check
  CHECK (purpose IN ('email_verify','password_reset','mfa_challenge'));
```

```ts
/**
 * Reads a live token without claiming it. Needed by the MFA challenge, where the code is
 * checked before the challenge is spent: consuming first would mean one mistyped digit
 * costs the user their challenge and sends them back through the password.
 *
 * Returns null for unknown, expired, consumed and wrong-purpose alike, exactly as
 * consumeAuthToken does — the caller must not tell them apart.
 */
export async function peekAuthToken(
  db: Db,
  tokenHash: string,
  purpose: AuthTokenPurpose,
): Promise<{ userId: string } | null> {
  const r = await db.query(
    `SELECT user_id FROM auth_tokens
      WHERE token_hash = $1 AND purpose = $2 AND consumed_at IS NULL AND expires_at > now()`,
    [tokenHash, purpose],
```

```ts
import { type Db } from '../pool.js';

export interface MfaRow {
  userId: string;
  secretEnc: Buffer;
  rootKeyId: string;
  confirmedAt: string | null;
  /** BIGINT: node-postgres hands it back as a string, so it is only ever compared in SQL. */
  lastUsedStep: string | null;
}

interface MfaDbRow {
  user_id: string;
  secret_enc: Buffer;
  root_key_id: string;
  confirmed_at: string | null;
  last_used_step: string | null;
}

/**
 * Stores a fresh unconfirmed secret. Returns false — without writing — when a confirmed
 * factor already exists, so re-running setup cannot swap the secret out from under a
 * factor the user is still relying on.
 */
export async function upsertPendingMfa(
  db: Db,
  args: { userId: string; secretEnc: Buffer; rootKeyId: string },
): Promise<boolean> {
  const r = await db.query(
    `INSERT INTO user_mfa (user_id, secret_enc, root_key_id) VALUES ($1,$2,$3)
       ON CONFLICT (user_id) DO UPDATE
         SET secret_enc = EXCLUDED.secret_enc,
             root_key_id = EXCLUDED.root_key_id,
             last_used_step = NULL,
             created_at = now()
       WHERE user_mfa.confirmed_at IS NULL`,
    [args.userId, args.secretEnc, args.rootKeyId],
  );
  return (r.rowCount ?? 0) > 0;
}

export async function getMfa(db: Db, userId: string): Promise<MfaRow | null> {
  const r = await db.query(`SELECT * FROM user_mfa WHERE user_id = $1`, [userId]);
  if (!r.rowCount) return null;
  const row = r.rows[0] as MfaDbRow;
  return {
    userId: row.user_id,
    secretEnc: row.secret_enc,
    rootKeyId: row.root_key_id,
    confirmedAt: row.confirmed_at,
    lastUsedStep: row.last_used_step,
  };
}

/** Idempotent: false means it was already confirmed, not that anything failed. */
export async function confirmMfa(db: Db, userId: string): Promise<boolean> {
  const r = await db.query(`UPDATE user_mfa SET confirmed_at = now() WHERE user_id = $1 AND confirmed_at IS NULL`, [
    userId,
  ]);
  return (r.rowCount ?? 0) > 0;
}

export async function deleteMfa(db: Db, userId: string): Promise<void> {
  await db.query(`DELETE FROM user_mfa WHERE user_id = $1`, [userId]);
  await db.query(`DELETE FROM mfa_recovery_codes WHERE user_id = $1`, [userId]);
}

export async function hasConfirmedMfa(db: Db, userId: string): Promise<boolean> {
  const r = await db.query(`SELECT 1 FROM user_mfa WHERE user_id = $1 AND confirmed_at IS NOT NULL`, [userId]);
  return (r.rowCount ?? 0) > 0;
}

/**
 * Compare-and-set on the replay floor: one statement both checks that the step is newer
 * than anything already spent and records it, so two concurrent logins presenting the same
 * code cannot both win.
 */
export async function claimTotpStep(db: Db, userId: string, step: number): Promise<boolean> {
  const r = await db.query(
    `UPDATE user_mfa SET last_used_step = $2
      WHERE user_id = $1 AND (last_used_step IS NULL OR last_used_step < $2)`,
    [userId, step],
  );
  return (r.rowCount ?? 0) > 0;
}

/** Replaces the entire set: any code from the previous batch stops working. */
export async function replaceRecoveryCodes(db: Db, userId: string, hashes: string[]): Promise<void> {
  await db.query(`DELETE FROM mfa_recovery_codes WHERE user_id = $1`, [userId]);
  for (const hash of hashes) {
    await db.query(`INSERT INTO mfa_recovery_codes (user_id, code_hash) VALUES ($1,$2)`, [userId, hash]);
  }
}

/** Compare-and-set, scoped to the owner so one user's code cannot be spent by another. */
export async function consumeRecoveryCode(db: Db, userId: string, hash: string): Promise<boolean> {
  const r = await db.query(
    `UPDATE mfa_recovery_codes SET consumed_at = now()
      WHERE user_id = $1 AND code_hash = $2 AND consumed_at IS NULL`,
    [userId, hash],
  );
  return (r.rowCount ?? 0) > 0;
}

export async function countLiveRecoveryCodes(db: Db, userId: string): Promise<number> {
  const r = await db.query(`SELECT count(*)::int AS c FROM mfa_recovery_codes WHERE user_id = $1 AND consumed_at IS NULL`, [
    userId,
  ]);
  return (r.rows[0] as { c: number }).c;
}
```

```ts
import {
  appendAuditEvent,
  claimTotpStep,
  confirmMfa,
  countLiveRecoveryCodes,
  generateToken,
  generateTotpSecret,
  getMfa,
  getUserById,
  hashToken,
  otpauthUri,
  replaceRecoveryCodes,
  upsertPendingMfa,
  verifyTotpCode,
  type MfaRow,
} from '@xperiment/db';
import type { FastifyInstance, FastifyReply } from 'fastify';
import { z } from 'zod';
import { auditActor, authOf } from '../auth/plugin.js';
import { sendProblem } from '../problem.js';
import type { ResolvedServerDeps } from './deps.js';

const ISSUER = 'Xperiment';
const RECOVERY_CODE_COUNT = 10;
const codeSchema = z.object({ code: z.string().min(6).max(64) }).strict();

/** Ten short opaque codes. The raw values leave the process once, in the response body. */
function mintRecoveryCodes(): { codes: string[]; hashes: string[] } {
  const codes = Array.from({ length: RECOVERY_CODE_COUNT }, () => generateToken('qm').token.slice(3, 17));
  return { codes, hashes: codes.map(hashToken) };
}

export function registerMfaRoutes(app: FastifyInstance, deps: ResolvedServerDeps): void {
  /**
   * Unwraps the stored secret. A row wrapped by a different root key cannot be opened, so
   * say so plainly rather than letting every correct code look wrong.
   */
  async function openSecret(row: MfaRow, reply: FastifyReply): Promise<Buffer | null> {
    if (row.rootKeyId !== deps.keyProvider.keyId) {
      sendProblem(reply, 503, 'MFA temporarily unavailable', 'This account was enrolled under a different root key');
      return null;
    }
    return deps.keyProvider.unwrap(row.secretEnc);
  }

  app.post('/v1/auth/mfa/setup', async (req, reply) => {
    const ctx = authOf(req);
    if (!ctx.userId) return sendProblem(reply, 400, 'Sessions only', 'API keys have no second factor');
    const user = await getUserById(deps.pool, ctx.userId);
    if (!user) return sendProblem(reply, 400, 'Sessions only');
    const { secret, base32 } = generateTotpSecret();
    const stored = await upsertPendingMfa(deps.pool, {
      userId: ctx.userId,
      secretEnc: await deps.keyProvider.wrap(secret),
      rootKeyId: deps.keyProvider.keyId,
    });
    // A confirmed factor is never silently replaced: disable it first, deliberately.
    if (!stored) return sendProblem(reply, 409, 'Already enrolled', 'Disable the current factor first');
    return { secretBase32: base32, otpauthUri: otpauthUri({ issuer: ISSUER, account: user.email, base32 }) };
  });

  app.post('/v1/auth/mfa/confirm', async (req, reply) => {
    const ctx = authOf(req);
    if (!ctx.userId) return sendProblem(reply, 400, 'Sessions only', 'API keys have no second factor');
    const parsed = codeSchema.safeParse(req.body ?? {});
    if (!parsed.success) return sendProblem(reply, 400, 'Invalid code');
    const row = await getMfa(deps.pool, ctx.userId);
    if (!row || row.confirmedAt) return sendProblem(reply, 400, 'Invalid code');
    const secret = await openSecret(row, reply);
    if (!secret) return reply;
    const step = verifyTotpCode({ secret, code: parsed.data.code, after: Number(row.lastUsedStep ?? 0) || null });
    if (step === null) return sendProblem(reply, 400, 'Invalid code');
    await claimTotpStep(deps.pool, ctx.userId, step);
    await confirmMfa(deps.pool, ctx.userId);
    const { codes, hashes } = mintRecoveryCodes();
    await replaceRecoveryCodes(deps.pool, ctx.userId, hashes);
    await appendAuditEvent(deps.pool, {
      ...auditActor(req),
      action: 'auth.mfa_enabled',
      targetType: 'user',
      targetId: ctx.userId,
    });
    // The only time these leave the server. There is no endpoint that shows them again.
    return { recoveryCodes: codes, remaining: await countLiveRecoveryCodes(deps.pool, ctx.userId) };
  });
}
```

```ts
  app.post('/v1/auth/mfa/verify', async (req, reply) => {
    const parsed = z
      .object({ challenge: z.string().min(1).max(200), code: z.string().min(6).max(64) })
      .strict()
      .safeParse(req.body ?? {});
    if (!parsed.success) return sendProblem(reply, 400, 'Invalid or expired token');
    const challengeHash = hashToken(parsed.data.challenge);
    if (!verifyLimiter.check(`mfa:${challengeHash}`)) {
      // Out of attempts: destroy the challenge rather than leave it sitting there for the
      // next minute's worth of guesses.
      await consumeAuthToken(deps.pool, challengeHash, 'mfa_challenge');
      return sendProblem(reply, 429, 'Too many attempts', 'Sign in again');
    }
    // Peek, not consume: a wrong code must not cost the user their challenge.
    const peeked = await peekAuthToken(deps.pool, challengeHash, 'mfa_challenge');
    if (!peeked) return sendProblem(reply, 400, 'Invalid or expired token');
    const kind = await proveCode(peeked.userId, parsed.data.code, reply);
    if (reply.sent) return reply;
    if (!kind) {
      await appendAuditEvent(deps.pool, {
        ...requestFacts(req),
        actorKind: 'user',
        actorUserId: peeked.userId,
        action: 'auth.mfa_failed',
        targetType: 'user',
        targetId: peeked.userId,
        outcome: 'failure',
      });
      return sendProblem(reply, 400, 'Invalid code');
    }
    // Only now is the challenge spent, and the CAS decides a race between two winners.
    const claimed = await consumeAuthToken(deps.pool, challengeHash, 'mfa_challenge');
    if (!claimed) return sendProblem(reply, 400, 'Invalid or expired token');
    verifyLimiter.reset(`mfa:${challengeHash}`);
    const { token, hash } = generateToken('qs');
    await createSession(deps.pool, {
      userId: peeked.userId,
      tokenHash: hash,
      expiresAt: new Date(Date.now() + SESSION_TTL_MS),
    });
    const facts = { ...requestFacts(req), actorKind: 'user' as const, actorUserId: peeked.userId };
    await appendAuditEvent(deps.pool, {
      ...facts,
      action: kind === 'recovery' ? 'auth.mfa_recovery_used' : 'auth.mfa_verified',
      targetType: 'user',
      targetId: peeked.userId,
    });
    await appendAuditEvent(deps.pool, {
      ...facts,
      action: 'auth.login_succeeded',
      targetType: 'user',
      targetId: peeked.userId,
    });
    return { token, recoveryCodesRemaining: await countLiveRecoveryCodes(deps.pool, peeked.userId) };
  });
```

```ts
/**
 * Routes that stay reachable while a workspace's MFA policy is unmet. This is a SEPARATE
 * list from EXEMPT: EXEMPT decides who needs a session at all, this decides which
 * authenticated routes survive an unmet policy. Without it the policy makes the workspace
 * permanently unusable — the dead-route trap from 3O and AUTH-002.
 */
const MFA_FIX_PATHS = new Set(['/v1/auth/me', '/v1/auth/logout']);
const MFA_FIX_PREFIX = '/v1/auth/mfa/';
const PRIVILEGED: ReadonlySet<Role> = new Set<Role>(['owner', 'admin']);

async function mfaPolicyBlocks(
  pool: pg.Pool,
  path: string,
  ctx: AuthContext,
  requireMfa: boolean,

  if (MFA_FIX_PATHS.has(path) || path.startsWith(MFA_FIX_PREFIX)) return false;
  return !(await hasConfirmedMfa(pool, ctx.userId));
}
```

```tsx
'use client';

import { useEffect, useState } from 'react';

/**
 * Renders the enrolment URI as a QR the user scans. The library is imported dynamically so
 * it never reaches the initial bundle, and the encoding happens in the browser — the secret
 * is not handed to any third-party service to draw.
 */
export function QrCode({ value, size = 192 }: { value: string; size?: number }) {
  const [dataUrl, setDataUrl] = useState<string | null>(null);
  const [failed, setFailed] = useState(false);

  useEffect(() => {
    let alive = true;
    void import('qrcode')
      .then((m) => m.toDataURL(value, { width: size, margin: 1 }))
      .then((url) => {
        if (alive) setDataUrl(url);
      })
      .catch(() => {
        if (alive) setFailed(true);
      });
    return () => {
      alive = false;
    };
  }, [value, size]);

  if (failed) return <p data-testid="qr-failed">Could not draw the code — use the key below instead.</p>;
  if (!dataUrl) return <div style={{ width: size, height: size }} aria-hidden />;
  return (
    // eslint-disable-next-line @next/next/no-img-element -- a data: URI, nothing to optimise
    <img src={dataUrl} width={size} height={size} alt="Scan this with your authenticator app" data-testid="mfa-qr" />
  );
}
```

```tsx
      setError(e instanceof ApiError ? e.message : 'Something went wrong');

      setBusy(false);
```

Add to `README.md`, immediately after the "Account recovery (AUTH-002)" section:

```markdown
### Two-factor authentication (AUTH-004)

Any user may enable a TOTP second factor from `/security`. The secret is wrapped by the
deployment **root key** rather than a workspace KEK, because MFA is account-level: filing it
under one workspace would mean destroying that workspace's key locks the user out of every
other workspace they belong to. Under `KMS_PROVIDER=gcp-kms` that means **one KMS decrypt per
MFA login** — human login rates, not request rates.

Login becomes two steps once a factor is confirmed: the password answers with a five-minute
challenge instead of a session, and `POST /v1/auth/mfa/verify` trades that challenge plus a
code for the session. A wrong code does not burn the challenge; running out of attempts
does. **A password reset does not bypass the second factor.**

Codes are single use: the verification window is ±1 step, so a code is live for 90 seconds,
and `user_mfa.last_used_step` is the floor that stops that window from doubling as a replay
window.

Ten single-use recovery codes are issued at enrolment and shown **once**. They are the only
way back in without the authenticator — an owner cannot reset a member's factor.

An owner may require the factor of every owner and admin in a workspace
(`PUT /v1/workspaces/mfa-policy`). Turning it on requires the owner to be enrolled already.
While the policy is unmet, a privileged member gets `403 MFA required` on everything except
`/v1/auth/me`, `/v1/auth/mfa/*` and `/v1/auth/logout`, so the account can always be fixed.

**The policy does not cover API keys.** A key is not a human and cannot present a code, so
enforcing it against keys would break every running automation the moment the switch is
flipped. A key carrying the admin role keeps full access under the policy; revoking the key
is the control that applies.
```

```ts
  await loginWithToken(page);
  await page.getByTestId('nav-security').click();
  await expect(page.getByTestId('security-page')).toBeVisible();
  await page.getByTestId('mfa-start').click();
  // The QR is drawn in the browser; the key beside it is what this test enrols with.
  await expect(page.getByTestId('mfa-qr')).toBeVisible();
  const key = (await page.getByTestId('mfa-key').innerText()).replace(/\s/g, '');

  const code = await page.evaluate(async (base32: string) => {
    // Recomputed in the page rather than imported: e2e runs against the built app.
    const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
    let bits = 0;
    let value = 0;
    const bytes: number[] = [];
    for (const ch of base32) {
      value = (value << 5) | alphabet.indexOf(ch);
      bits += 5;
      if (bits >= 8) {
        bytes.push((value >>> (bits - 8)) & 255);
        bits -= 8;
      }
    }
    const secret = new Uint8Array(bytes);
    const counter = Math.floor(Date.now() / 1000 / 30);
    const msg = new ArrayBuffer(8);
    new DataView(msg).setBigUint64(0, BigInt(counter));
    const key = await crypto.subtle.importKey('raw', secret, { name: 'HMAC', hash: 'SHA-1' }, false, ['sign']);
    const mac = new Uint8Array(await crypto.subtle.sign('HMAC', key, msg));
    const offset = mac[mac.length - 1]! & 0x0f;
    const bin =
      ((mac[offset]! & 0x7f) << 24) | ((mac[offset + 1]! & 0xff) << 16) | ((mac[offset + 2]! & 0xff) << 8) | mac[offset + 3]!;
    return String(bin % 1000000).padStart(6, '0');
  }, key);

  await page.getByTestId('mfa-code').fill(code);
  await page.getByTestId('mfa-confirm').click();
  await expect(page.getByTestId('mfa-recovery')).toBeVisible();
  await page.screenshot({ path: `${ARTIFACTS}/10-mfa.png` });
});
```
