# Team RBAC & Granular Permissions (Phase 3 Sub-project M) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the linear role ladder with a capability model so Operator (runtime without authoring) and Billing (orthogonal) are expressible, and every route declares the permission it needs.

**Architecture:** A static `Permission` catalog + `ROLE_PERMISSIONS` map in `apps/api/src/auth/permissions.ts`; `requirePermission` replaces `roleGate`/`requireRole` (both **removed**) at all 25 call sites; `api_keys` carry a role; member list/role-change routes with last-owner and owner-grant guards; a web Members page.

**Tech Stack:** pnpm monorepo, TS ESM via tsx, Vitest (v8 coverage ≥80%), PostgreSQL (pg), Fastify, Next.js (Arus).

## Global Constraints

- No `console.log`; no hardcoded secrets. TDD: failing test first. Coverage ≥80% every metric per package.
- Conventional commits `<type>: <description>`, NO attribution trailer. Immutability; focused files.
- The six roles are exactly `'viewer' | 'builder' | 'operator' | 'billing' | 'admin' | 'owner'`.
- `roleGate` and `requireRole` are DELETED — no compatibility shim. Removing them breaks compilation until every call site migrates, which is intentional (Task M-2 is atomic).
- 403 messages name the **permission**, never a role.
- Fail closed: a role missing from `ROLE_PERMISSIONS` has no permissions and must not throw.
- Migrations auto-apply by sorted filename; next is `0009_rbac.sql`.
- Reuse: `authOf(req)`, `sendProblem(reply, status, title, detail)`, `createAuthedUser` (apps/api/test/authHelpers.ts).
- Branch `feature/phase3-rbac` is checked out; the spec is committed. Docker Postgres/Redis running.

```ts
    name: row['name'] as string,
    role: row['role'] as Role,
    lastUsedAt: ts(row['last_used_at']),
    createdAt: ts(row['created_at'])!,
  };
}
```

```ts
export async function listMembers(
  db: Db,
  workspaceId: string,
): Promise<{ userId: string; email: string; role: Role; createdAt: string }[]> {
  const r = await db.query(
    `SELECT m.user_id, u.email, m.role, m.created_at
     FROM workspace_members m JOIN users u ON u.id = m.user_id
     WHERE m.workspace_id = $1 ORDER BY m.created_at`,
    [workspaceId],
  );
  const ts = (v: unknown) => (v instanceof Date ? v.toISOString() : (v as string));
  return (r.rows as Record<string, unknown>[]).map((row) => ({
    userId: row['user_id'] as string,
    email: row['email'] as string,
    role: row['role'] as Role,
    createdAt: ts(row['created_at']),
  }));
}

/**
 * Change a member's role. Refuses to demote the last owner — checked inside the
 * same transaction as the update so two concurrent demotions cannot both win.
 */
export async function updateMemberRole(
  pool: pg.Pool,
  args: { workspaceId: string; userId: string; role: Role },
): Promise<'ok' | 'not_member' | 'last_owner'> {
  return withTx(pool, async (client) => {
    const current = await client.query(
      `SELECT role FROM workspace_members WHERE workspace_id = $1 AND user_id = $2 FOR UPDATE`,
      [args.workspaceId, args.userId],
    );
    if (!current.rowCount) return 'not_member';
    const was = (current.rows[0] as { role: Role }).role;
    if (was === 'owner' && args.role !== 'owner') {
      const owners = await client.query(
        `SELECT count(*)::int AS c FROM workspace_members WHERE workspace_id = $1 AND role = 'owner'`,
        [args.workspaceId],
      );
      if ((owners.rows[0] as { c: number }).c <= 1) return 'last_owner';
    }
    await client.query(`UPDATE workspace_members SET role = $3 WHERE workspace_id = $1 AND user_id = $2`, [
      args.workspaceId,
      args.userId,
      args.role,
    ]);
    return 'ok';
  });
}
```

```ts
import type { Role } from '@xperiment/db';

export type Permission =
  | 'workflow:read'
  | 'workflow:write'
  | 'workflow:activate'
  | 'execution:read'
  | 'execution:run'
  | 'execution:cancel'
  | 'approval:decide'
  | 'connection:read'
  | 'connection:write'
  | 'alert:read'
  | 'alert:write'
  | 'usage:read'
  | 'billing:read'
  | 'billing:manage'
  | 'member:manage';

const READ_ONLY: Permission[] = ['workflow:read', 'execution:read', 'alert:read', 'usage:read'];
const RUNTIME: Permission[] = ['execution:run', 'execution:cancel', 'approval:decide', 'alert:write'];
const AUTHORING: Permission[] = ['workflow:write', 'workflow:activate', 'connection:read', 'connection:write'];
const BILLING: Permission[] = ['billing:read', 'billing:manage'];

const OPERATOR: Permission[] = [...READ_ONLY, ...RUNTIME];
const BUILDER: Permission[] = [...OPERATOR, ...AUTHORING];
const ADMIN: Permission[] = [...BUILDER, ...BILLING, 'member:manage'];

/**
 * Built-in role → permission sets. Deliberately a plain data structure: WS-006's
 * future custom roles can replace the source of this map without touching any route.
 */
export const ROLE_PERMISSIONS: Record<Role, ReadonlySet<Permission>> = {
  viewer: new Set(READ_ONLY),
  operator: new Set(OPERATOR),
  billing: new Set(['alert:read', 'usage:read', ...BILLING]),
  builder: new Set(BUILDER),
  admin: new Set(ADMIN),
  owner: new Set(ADMIN),
};

/** Fails closed: an unknown role has no permissions and never throws. */
export function hasPermission(role: Role, permission: Permission): boolean {
  return ROLE_PERMISSIONS[role]?.has(permission) ?? false;
}
```

```ts
/**
 * Registers a fresh user and adds them to `workspaceId` with `role`, returning an
 * inject bound to that workspace (via X-Xperiment-Workspace) so the caller's context
 * carries the requested role.
 */
export async function addMemberWithRole(
  app: FastifyInstance,
  pool: import('pg').Pool,
  workspaceId: string,
  role: string,
): Promise<AuthedUser> {
  const user = await createAuthedUser(app);
  await pool.query(
    `INSERT INTO workspace_members (workspace_id, user_id, role) VALUES ($1,$2,$3)
     ON CONFLICT (workspace_id, user_id) DO UPDATE SET role = EXCLUDED.role`,
    [workspaceId, user.userId, role],
  );
  return {
    ...user,
    workspaceId,
    inject: (opts: InjectOptions) =>
      app.inject({
        ...opts,
        headers: {
          authorization: `Bearer ${user.token}`,
          'x-xperiment-workspace': workspaceId,
          ...(opts.headers ?? {}),
        },
      } as InjectOptions) as Promise<InjectResponse>,
  };
}
```

```ts
    const { userId } = req.params as { userId: string };
```
